$$
\int x^kdx=\frac1{k+1}x^{k+1}+C \qquad
\int \frac1{x^2}dx=-\frac1{x}+C\qquad \int \frac1{\sqrt{x}}dx=2\sqrt{x}+C\qquad
\int \frac1{x}dx=\ln{|x|}+C
$$
$$
\int e^xdx=e^x+C\qquad \int a^xdx=\frac{a^x}{\ln{a}}+C\ (a>0,a\ne1)
$$
---
$$
\int\sin{x}dx=-\cos{x}+C\qquad \int\cos{x}dx=\sin{x}+C
$$
$$
\int\tan{x}dx=-\ln{|\cos{x}|}+C\qquad \int \cot{x}dx=\ln{|\sin{x}|}+C
$$
---
$$
!!\quad\int\frac1{\cos{x}}dx=\int\sec{x}dx=\ln{|\sec{x}+\tan{x}|}+C\quad !!
$$
$$
\int \frac1{\sin{x}}dx=\int\csc{x}dx=\ln{|\csc{x}-\cot{x}|}+C
$$
---
$$
\int \sec^2{x}dx=\tan{x}+C\qquad \int\csc^2{x}dx=-\cot{x}+C
$$
$$
\int\sec{x}\tan{x}dx=\sec{x}+C\qquad \int\csc{x}\cot{x}dx=-\csc{x}+C
$$
---
$$
\int\frac1{1+x^2}dx=\arctan{x}+C\qquad \int\frac1{a^2+x^2}dx=\frac1{a}\arctan{\frac{x}{a}}+C\ (a>0)
$$
$$
\int\frac1{\sqrt{x^2+a^2}}dx=\ln{(x+\sqrt{x^2+a^2})}+C\qquad
\int\frac1{\sqrt{x^2-a^2}}dx=\ln{(x+\sqrt{x^2-a^2})}+C\ (|x|>|a|)
$$
---
$$
\int\frac1{x^2-a^2}dx=\frac1{2a}\ln{\left|\frac{x-a}{x+a}\right|}+C 
$$
---
$$
!!\quad\int\sqrt{a^2-x^2}dx=\frac{a^2}{2}\arcsin{\frac{x}{a}}+\frac{x}{2}\sqrt{a^2-x^2}+C\ (a>|x|\ge0)\quad !!
$$

$$
\int \sin^2 x \, dx = \frac{x}{2} - \frac{\sin 2x}{4} + C
\qquad
(\sin^2 x = \frac{1 - \cos 2x}{2})
$$
$$
\int \cos^2 x \, dx = \frac{x}{2} + \frac{\sin 2x}{4} + C
\qquad
(\cos^2 x = \frac{1 + \cos 2x}{2})
$$
$$
\int \tan^2 x \, dx = \tan x - x + C
\qquad
(\tan^2 x = \sec^2 x - 1)
$$
$$
\int \cot^2 x \, dx = -\cot x - x + C
\qquad
(\cot^2 x = \csc^2 x - 1)
$$
